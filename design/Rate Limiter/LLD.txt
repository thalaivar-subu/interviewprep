public class RateLimiterService {
    private final Map<UserTier, RateLimiter> rateLimiters = new HashMap<>();

    public RateLimterService(){
        rateLimiters.put(UserTier.Free, RateLimiterFactory.createRateLimiter(RateLimiterType.TokenBucket, new RateLimitConfig(maxRequest=10, windowInSeconds= 60)))
    }

    public boolean allowRequest(User user){
        RateLimter rateLimiter = this.rateLimiters.get(user.getTier());
        if(rateLimiter== null){
            throw new IllegalArgumetnException
        }
        return rateLimiter.allowRequest(user.getUserId())
    }
}
